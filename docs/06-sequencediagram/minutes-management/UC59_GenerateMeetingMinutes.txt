title UC-59 - Generate Meeting Minutes

actor "Chair or Secretary" as Actor
boundary "Meeting Management Page" as MeetingPage
control ":Minutes Controller" as MinutesController
control ":Minutes Service" as MinutesService
participant ":Meeting Access Policy" as MeetingAccessPolicy
participant ":Minutes Repository" as MinutesRepository
database ":Database" as Database

Actor->MeetingPage: Select Create draft or Rebuild
activate MeetingPage

MeetingPage->MinutesController: Generate meeting minutes
activate MinutesController
MinutesController->MinutesService: generateMinutes(meetingId, actorId)
activate MinutesService

MinutesService->MeetingAccessPolicy: Check actor can access the Meeting
activate MeetingAccessPolicy
MeetingAccessPolicy->Database: Check Meeting relationship
activate Database
Database-->MeetingAccessPolicy: Access / No access
deactivate Database
MeetingAccessPolicy-->MinutesService: Access / No access
deactivate MeetingAccessPolicy

alt Actor is not the current Chair or Secretary - EX-1
    MinutesService-->MinutesController: MSG397 - Only the Chair or Secretary may write the minutes.
    MinutesController-->MeetingPage: MSG397 - Only the Chair or Secretary may write the minutes.
    MeetingPage-->Actor: MSG397 - Only the Chair or Secretary may write the minutes.
else Actor is Chair or Secretary
    alt Meeting has not ended - EX-2
        MinutesService-->MinutesController: MSG398 - The minutes can be written only after the meeting has ended.
        MinutesController-->MeetingPage: MSG398 - The minutes can be written only after the meeting has ended.
        MeetingPage-->Actor: MSG398 - The minutes can be written only after the meeting has ended.
    else Meeting has ended
        alt A draft already exists - EX-3
            MinutesService-->MinutesController: MSG399 - This meeting already has a minutes draft.
            MinutesController-->MeetingPage: MSG399 - This meeting already has a minutes draft.
            MeetingPage-->Actor: MSG399 - This meeting already has a minutes draft.
        else No draft exists
            MinutesService->MinutesRepository: Save draft minutes
            activate MinutesRepository
            MinutesRepository->Database: Insert minutes draft
            activate Database
            Database-->MinutesRepository: Saved
            deactivate Database
            MinutesRepository-->MinutesService: Saved draft
            deactivate MinutesRepository

            MinutesService-->MinutesController: MSG395 - Minutes draft created.
            MinutesController-->MeetingPage: MSG395 - Minutes draft created.
            MeetingPage-->Actor: MSG395 - Minutes draft created.
        end
    end
end

deactivate MinutesService
deactivate MinutesController
deactivate MeetingPage
