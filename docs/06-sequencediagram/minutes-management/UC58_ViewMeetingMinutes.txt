title UC-58 - View Meeting Minutes

actor "Chair / Secretary / Reader" as Actor
boundary "Minutes Card (Biên bản tab)" as MinutesCard
control ":Minutes Endpoints" as MinutesEndpoints
control ":Minutes Service" as MinutesService
participant ":Meeting Repository" as MeetingRepository
database ":Database" as Database

Actor->MinutesCard: Open the Biên bản tab, or choose a revision from the list
activate MinutesCard
MinutesCard->MinutesEndpoints: get(meetingId), or getRevision(minutesId)
activate MinutesEndpoints
MinutesEndpoints->MinutesService: getAsync(meetingId, actorId), or getRevisionAsync(minutesId, actorId)
activate MinutesService
MinutesService->MeetingRepository: Get the Meeting
activate MeetingRepository
MeetingRepository->Database: Retrieve Meeting
activate Database
Database-->MeetingRepository: Meeting / Not found
deactivate Database
MeetingRepository-->MinutesService: Meeting / Not found
deactivate MeetingRepository

alt The Meeting is not accessible - EX-3
    MinutesService-->MinutesEndpoints: Not found
    MinutesEndpoints-->MinutesCard: 404
    MinutesCard-->Actor: The Meeting is not available
else Actor may read the Meeting
    MinutesService->MinutesService: Check whether the actor is the current Chair or Secretary (BR-386)

    alt Actor is the current Chair or Secretary
        MinutesService->Database: Retrieve the latest revision, or the chosen revision, with\nwho created, submitted, approved and returned it, when, and the reason\nof the latest return (BR-407)
        activate Database
        Database-->MinutesService: Minutes / None
        deactivate Database

        alt A revision exists - the draft or returned revision, or a waiting one
            MinutesService-->MinutesEndpoints: 200 MinutesState with the revision and every revision in history
            MinutesEndpoints-->MinutesCard: 200 MinutesState
            MinutesCard-->Actor: The paper as it will print, revision number and status,\nwho created, submitted, approved or returned it and when,\nand the reason of the latest return
        else No minutes yet - AF-1
            MinutesService-->MinutesEndpoints: 200 MinutesState with no revision
            MinutesEndpoints-->MinutesCard: 200 MinutesState
            MinutesCard-->Actor: Empty state and the control to build one (UC-59)
        end
    else Actor is anyone else who may read the Meeting
        MinutesService->Database: Retrieve the latest Issued revision, and the revisions\nin history that are Issued or Superseded only (BR-412)
        activate Database
        Database-->MinutesService: Minutes / None, filtered history
        deactivate Database

        alt An Issued revision exists
            MinutesService-->MinutesEndpoints: 200 MinutesState with the Issued revision\nand the filtered history
            MinutesEndpoints-->MinutesCard: 200 MinutesState
            MinutesCard-->Actor: The issued paper, its revision number and status,\nand the revisions it replaced marked Superseded
        else No Issued revision yet - AF-1
            MinutesService-->MinutesEndpoints: 200 MinutesState with no revision
            MinutesEndpoints-->MinutesCard: 200 MinutesState
            MinutesCard-->Actor: No issued minutes yet; no draft or unissued correction is disclosed
        end
    end

    opt Actor asks for another revision - AF-2
        MinutesCard->MinutesEndpoints: getRevision(minutesId)
        MinutesEndpoints->MinutesService: getRevisionAsync(minutesId, actorId)
        alt The actor may read that revision - Issued or Superseded
            MinutesService-->MinutesEndpoints: 200 the chosen revision with the same state
            MinutesEndpoints-->MinutesCard: 200 MinutesState
            MinutesCard-->Actor: The chosen revision, with its own number and status
        else The revision is a draft, waiting, returned or an unissued correction - EX-2
            MinutesService-->MinutesEndpoints: Not found, no code
            MinutesEndpoints-->MinutesCard: 404
            MinutesCard-->Actor: No minutes; nothing about that revision is disclosed
        end
    end

    opt Actor downloads an Issued or Superseded revision - AF-3
        MinutesCard->MinutesEndpoints: export(minutesId, format)
        MinutesEndpoints->MinutesService: exportAsync(minutesId, actorId, format)
        alt The Meeting is Secret - EX-4
            MinutesService-->MinutesEndpoints: MSG405 - The minutes of a Secret meeting cannot be downloaded.
            MinutesEndpoints-->MinutesCard: 403 MSG405
            MinutesCard-->Actor: MSG405 - The minutes of a Secret meeting cannot be downloaded.
        else The revision is a draft, waiting, returned or an unissued correction - EX-2
            MinutesEndpoints->MinutesService: exportAsync(minutesId, actorId, format)
            MinutesService-->MinutesEndpoints: Not found
            MinutesEndpoints-->MinutesCard: 404
            MinutesCard-->Actor: No minutes; nothing about that revision is disclosed
        else Not Secret and the actor may read the revision
            MinutesService-->MinutesEndpoints: 200 file built from the same content (BR-390, BR-391, BR-412)
            MinutesEndpoints-->MinutesCard: 200 .docx / .pdf
            MinutesCard-->Actor: The file
        end
    end
end
deactivate MinutesService

opt The minutes cannot be read - EX-1
    MinutesService-->MinutesEndpoints: MSG403 - Unable to load the minutes.
    MinutesEndpoints-->MinutesCard: MSG403 - Unable to load the minutes.
    MinutesCard-->Actor: MSG403 - Unable to load the minutes.
end
deactivate MinutesEndpoints
deactivate MinutesCard
