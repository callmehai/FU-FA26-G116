title UC-61 - Submit Meeting Minutes for Approval

actor "Secretary" as Actor
boundary "Meeting Management Page" as MeetingPage
control ":Minutes Controller" as MinutesController
control ":Minutes Service" as MinutesService
participant ":Minutes Repository" as MinutesRepository
database ":Database" as Database

Actor->MeetingPage: Select Submit for approval
activate MeetingPage

MeetingPage->MinutesController: Submit meeting minutes for approval
activate MinutesController
MinutesController->MinutesService: submitMinutes(meetingId, actorId)
activate MinutesService

MinutesService->MinutesRepository: Load the revision
activate MinutesRepository
MinutesRepository->Database: Retrieve revision
activate Database
Database-->MinutesRepository: Revision
deactivate Database
MinutesRepository-->MinutesService: Revision
deactivate MinutesRepository

alt Actor is the current Chair, not the Secretary - EX-1
    MinutesService-->MinutesController: MSG425 - Only the Secretary may submit the minutes for approval.
    MinutesController-->MeetingPage: MSG425 - Only the Secretary may submit the minutes for approval.
    MeetingPage-->Actor: MSG425 - Only the Secretary may submit the minutes for approval.
else Actor is the Secretary
    alt Meeting has not ended - EX-2
        MinutesService-->MinutesController: MSG426 - The minutes can be submitted only after the meeting has ended.
        MinutesController-->MeetingPage: MSG426 - The minutes can be submitted only after the meeting has ended.
        MeetingPage-->Actor: MSG426 - The minutes can be submitted only after the meeting has ended.
    else Meeting has ended
        alt Revision was already submitted or issued - EX-3
            MinutesService-->MinutesController: MSG400 - The minutes have been submitted or issued and can no longer be changed.
            MinutesController-->MeetingPage: MSG400 - The minutes have been submitted or issued and can no longer be changed.
            MeetingPage-->Actor: MSG400 - The minutes have been submitted or issued and can no longer be changed.
        else Revision is a draft or returned
            MinutesService->MinutesRepository: Mark revision waiting for approval
            activate MinutesRepository
            MinutesRepository->Database: Update revision and notification
            activate Database
            Database-->MinutesRepository: Updated
            deactivate Database
            MinutesRepository-->MinutesService: Updated
            deactivate MinutesRepository

            MinutesService-->MinutesController: MSG424 - Minutes submitted to the Chair for approval.
            MinutesController-->MeetingPage: MSG424 - Minutes submitted to the Chair for approval.
            MeetingPage-->Actor: MSG424 - Minutes submitted to the Chair for approval.
        end
    end
end

deactivate MinutesService
deactivate MinutesController
deactivate MeetingPage
