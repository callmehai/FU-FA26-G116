title UC-71 - Edit Meeting Minutes

actor "Chair or Secretary" as Actor
boundary "Meeting Management Page" as MeetingPage
control ":Minutes Controller" as MinutesController
control ":Minutes Service" as MinutesService
participant ":Minutes Repository" as MinutesRepository
database ":Database" as Database

Actor->MeetingPage: Select Edit and Save
activate MeetingPage

MeetingPage->MinutesController: Edit meeting minutes
activate MinutesController
MinutesController->MinutesService: editMinutes(meetingId, actorId, content)
activate MinutesService

MinutesService->MinutesRepository: Load the revision
activate MinutesRepository
MinutesRepository->Database: Retrieve revision
activate Database
Database-->MinutesRepository: Revision
deactivate Database
MinutesRepository-->MinutesService: Revision
deactivate MinutesRepository

alt Someone else saved first - EX-1
    MinutesService-->MinutesController: MSG401 - The minutes were just changed by someone else. Reload to see the latest version.
    MinutesController-->MeetingPage: MSG401 - The minutes were just changed by someone else. Reload to see the latest version.
    MeetingPage-->Actor: MSG401 - The minutes were just changed by someone else. Reload to see the latest version.
else No concurrent save
    alt Content is invalid - EX-2
        MinutesService-->MinutesController: MSG402 - The minutes content is invalid.
        MinutesController-->MeetingPage: MSG402 - The minutes content is invalid.
        MeetingPage-->Actor: MSG402 - The minutes content is invalid.
    else Content is valid
        alt Minutes can no longer be changed - EX-3
            MinutesService-->MinutesController: MSG400 - The minutes have been submitted or issued and can no longer be changed.
            MinutesController-->MeetingPage: MSG400 - The minutes have been submitted or issued and can no longer be changed.
            MeetingPage-->Actor: MSG400 - The minutes have been submitted or issued and can no longer be changed.
        else Minutes can be changed
            MinutesService->MinutesRepository: Save content and save time
            activate MinutesRepository
            MinutesRepository->Database: Update minutes
            activate Database
            Database-->MinutesRepository: Saved
            deactivate Database
            MinutesRepository-->MinutesService: Saved
            deactivate MinutesRepository

            MinutesService-->MinutesController: MSG396 - Minutes saved.
            MinutesController-->MeetingPage: MSG396 - Minutes saved.
            MeetingPage-->Actor: MSG396 - Minutes saved.
        end
    end
end

deactivate MinutesService
deactivate MinutesController
deactivate MeetingPage
