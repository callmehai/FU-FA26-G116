title UC-33b - Attach Documents to Agenda Item

actor "Chair or Secretary" as Actor
boundary "Meeting Documents Tab" as DocumentsTab
control ":Meeting Document Controller" as DocumentController
control ":Meeting Document Service" as DocumentService
participant ":Meeting Access Policy" as MeetingAccessPolicy
participant ":Library Document Repository" as LibraryDocumentRepository
participant ":Meeting Document Repository" as MeetingDocumentRepository
participant ":Object Storage (R2)" as FileStorage
database ":Database" as Database

Actor->DocumentsTab: Submit the selected documents of the chosen source
activate DocumentsTab

DocumentsTab->DocumentController: Attach the documents to the Agenda Item
activate DocumentController

DocumentController->DocumentService: attachDocuments(meetingId, agendaItemId, source, documents, actor)
activate DocumentService

DocumentService->MeetingAccessPolicy: Check the actor's Meeting role and the Meeting status
activate MeetingAccessPolicy
MeetingAccessPolicy->Database: Retrieve Meeting status and the actor's role
activate Database
Database-->MeetingAccessPolicy: Meeting status and role
deactivate Database
MeetingAccessPolicy-->DocumentService: Role and Meeting status
deactivate MeetingAccessPolicy

alt Actor is no longer the Chair or Secretary - EX-1
    DocumentService-->DocumentController: MSG240 - You no longer have permission to link documents.
    DocumentController-->DocumentsTab: MSG240 - You no longer have permission to link documents.
    DocumentsTab-->Actor: MSG240 - You no longer have permission to link documents.
else Meeting no longer accepts document links - EX-3
    DocumentService-->DocumentController: MSG247 - Document links cannot be changed after the Meeting has ended.
    DocumentController-->DocumentsTab: MSG247 - Document links cannot be changed after the Meeting has ended.
    DocumentsTab-->Actor: MSG247 - Document links cannot be changed after the Meeting has ended.
else Actor may link documents
    DocumentService->MeetingDocumentRepository: Get the Agenda Item and the documents already linked
    activate MeetingDocumentRepository
    MeetingDocumentRepository->Database: Retrieve the Agenda Item and its links
    activate Database
    Database-->MeetingDocumentRepository: Agenda Item and links
    deactivate Database
    MeetingDocumentRepository-->DocumentService: Agenda Item and links
    deactivate MeetingDocumentRepository

    alt Source is Meeting documents - Normal Flow
        DocumentService->MeetingDocumentRepository: Get the selected Meeting documents
        activate MeetingDocumentRepository
        MeetingDocumentRepository->Database: Retrieve the selected documents
        activate Database
        Database-->MeetingDocumentRepository: Meeting documents
        deactivate Database
        MeetingDocumentRepository-->DocumentService: Meeting documents
        deactivate MeetingDocumentRepository

        alt Agenda Item or a document is no longer in the Meeting, or already linked - EX-2
            DocumentService-->DocumentController: MSG170 - This document or Agenda Item is no longer available.
            DocumentController-->DocumentsTab: MSG170 - This document or Agenda Item is no longer available.
            DocumentsTab-->Actor: MSG170 - This document or Agenda Item is no longer available.
        else Selection is valid
            DocumentService->MeetingDocumentRepository: Attach the documents to the Agenda Item
            activate MeetingDocumentRepository
            MeetingDocumentRepository->Database: Insert the links
            activate Database
            Database-->MeetingDocumentRepository: Saved
            deactivate Database
            MeetingDocumentRepository-->DocumentService: Saved
            deactivate MeetingDocumentRepository

            DocumentService-->DocumentController: Documents attached
            DocumentController-->DocumentsTab: Documents attached
            DocumentsTab-->Actor: MSG238 - Documents attached to the Agenda Item.
        end
    else Source is Upload from computer - AF-1
        alt Any file is invalid or over the size limit - EX-2
            DocumentService-->DocumentController: MSG164 - Select valid PDF or DOCX files, each no larger than 10 MB.
            DocumentController-->DocumentsTab: MSG164 - Select valid PDF or DOCX files, each no larger than 10 MB.
            DocumentsTab-->Actor: MSG164 - Select valid PDF or DOCX files, each no larger than 10 MB.
        else Every file is valid
            DocumentService->FileStorage: Store each file
            activate FileStorage
            FileStorage-->DocumentService: Files stored
            deactivate FileStorage

            DocumentService->MeetingDocumentRepository: Save the documents and attach them to the Agenda Item
            activate MeetingDocumentRepository
            MeetingDocumentRepository->Database: Insert the documents and the links
            activate Database
            Database-->MeetingDocumentRepository: Saved
            deactivate Database
            MeetingDocumentRepository-->DocumentService: Saved
            deactivate MeetingDocumentRepository

            DocumentService-->DocumentController: Documents attached
            DocumentController-->DocumentsTab: Documents attached
            DocumentsTab-->Actor: MSG238 - Documents attached to the Agenda Item.
        end
    else Source is My Documents - AF-2
        DocumentService->LibraryDocumentRepository: Get the selected library documents
        activate LibraryDocumentRepository
        LibraryDocumentRepository->Database: Retrieve the actor's selected documents
        activate Database
        Database-->LibraryDocumentRepository: Library documents
        deactivate Database
        LibraryDocumentRepository-->DocumentService: Library documents
        deactivate LibraryDocumentRepository

        alt Selection is empty or unavailable - EX-2
            DocumentService-->DocumentController: MSG249 - Select available documents from your personal library.
            DocumentController-->DocumentsTab: MSG249 - Select available documents from your personal library.
            DocumentsTab-->Actor: MSG249 - Select available documents from your personal library.
        else Selection is valid
            DocumentService->MeetingDocumentRepository: Add the documents to the Meeting and attach them to the Agenda Item
            activate MeetingDocumentRepository
            MeetingDocumentRepository->Database: Insert the Meeting documents and the links
            activate Database
            Database-->MeetingDocumentRepository: Saved
            deactivate Database
            MeetingDocumentRepository-->DocumentService: Saved
            deactivate MeetingDocumentRepository

            DocumentService-->DocumentController: Documents attached
            DocumentController-->DocumentsTab: Documents attached
            DocumentsTab-->Actor: MSG238 - Documents attached to the Agenda Item.
        end
    end
end

deactivate DocumentService
deactivate DocumentController
deactivate DocumentsTab
