title UC-36a - Add Document to Meeting

actor "Chair or Secretary" as Actor
boundary "Meeting Documents Tab" as DocumentsTab
control ":Meeting Document Controller" as DocumentController
control ":Meeting Document Service" as DocumentService
participant ":Meeting Access Policy" as MeetingAccessPolicy
participant ":Library Document Repository" as LibraryDocumentRepository
participant ":Meeting Document Repository" as MeetingDocumentRepository
database ":Database" as Database

Actor->DocumentsTab: Submit the selected library documents
activate DocumentsTab

DocumentsTab->DocumentController: Add the library documents
activate DocumentController

DocumentController->DocumentService: addLibraryDocuments(meetingId, libraryDocumentIds, actor)
activate DocumentService

DocumentService->MeetingAccessPolicy: Check the actor's Meeting role and the Meeting status
activate MeetingAccessPolicy
MeetingAccessPolicy->Database: Retrieve Meeting status and the actor's role
activate Database
Database-->MeetingAccessPolicy: Meeting status and role
deactivate Database
MeetingAccessPolicy-->DocumentService: Role and Meeting status
deactivate MeetingAccessPolicy

alt Actor is no longer the Chair or Secretary - EX-1
    DocumentService-->DocumentController: MSG162 - You no longer have permission to add documents.
    DocumentController-->DocumentsTab: MSG162 - You no longer have permission to add documents.
    DocumentsTab-->Actor: MSG162 - You no longer have permission to add documents.
else Meeting no longer accepts documents - EX-3
    DocumentService-->DocumentController: MSG172 - Documents cannot be added after the Meeting has ended.
    DocumentController-->DocumentsTab: MSG172 - Documents cannot be added after the Meeting has ended.
    DocumentsTab-->Actor: MSG172 - Documents cannot be added after the Meeting has ended.
else Actor may add documents
    DocumentService->LibraryDocumentRepository: Get the selected library documents
    activate LibraryDocumentRepository
    LibraryDocumentRepository->Database: Retrieve the actor's selected documents
    activate Database
    Database-->LibraryDocumentRepository: Library documents
    deactivate Database
    LibraryDocumentRepository-->DocumentService: Library documents
    deactivate LibraryDocumentRepository

    DocumentService->MeetingDocumentRepository: Get the documents already in the Meeting
    activate MeetingDocumentRepository
    MeetingDocumentRepository->Database: Retrieve the Meeting documents
    activate Database
    Database-->MeetingDocumentRepository: Meeting documents
    deactivate Database
    MeetingDocumentRepository-->DocumentService: Meeting documents
    deactivate MeetingDocumentRepository

    alt Selection is empty or unavailable - EX-2
        DocumentService-->DocumentController: MSG249 - Select available documents from your personal library.
        DocumentController-->DocumentsTab: MSG249 - Select available documents from your personal library.
        DocumentsTab-->Actor: MSG249 - Select available documents from your personal library.
    else A selected document is already in the Meeting - EX-2
        DocumentService-->DocumentController: MSG165 - One or more selected documents are already attached to the Meeting.
        DocumentController-->DocumentsTab: MSG165 - One or more selected documents are already attached to the Meeting.
        DocumentsTab-->Actor: MSG165 - One or more selected documents are already attached to the Meeting.
    else Selection is valid
        DocumentService->MeetingDocumentRepository: Add the documents to the Meeting
        activate MeetingDocumentRepository
        MeetingDocumentRepository->Database: Insert the Meeting documents
        activate Database
        Database-->MeetingDocumentRepository: Saved
        deactivate Database
        MeetingDocumentRepository-->DocumentService: Saved
        deactivate MeetingDocumentRepository

        DocumentService-->DocumentController: Number of documents added
        DocumentController-->DocumentsTab: Number of documents added
        DocumentsTab-->Actor: MSG161 - Documents added to the Meeting.
    end
end

deactivate DocumentService
deactivate DocumentController
deactivate DocumentsTab
