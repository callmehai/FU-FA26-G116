title UC-36 - Upload Document to Personal Library

actor "Internal User" as Actor
boundary "Personal Library Screen" as LibraryScreen
control ":Document Controller" as DocumentController
control ":Document Service" as DocumentService
participant ":Folder Repository" as FolderRepository
participant ":Document Repository" as DocumentRepository
participant ":Object Storage (R2)" as FileStorage
database ":Database" as Database

Actor->LibraryScreen: Submit the selected files
activate LibraryScreen

LibraryScreen->DocumentController: Upload the files
activate DocumentController

DocumentController->DocumentService: uploadDocuments(userId, folderId, files)
activate DocumentService

opt A destination folder is selected
    DocumentService->FolderRepository: Get the destination folder
    activate FolderRepository
    FolderRepository->Database: Retrieve the folder by id and owner
    activate Database
    Database-->FolderRepository: Folder / Not found
    deactivate Database
    FolderRepository-->DocumentService: Folder / Not found
    deactivate FolderRepository
end

alt Destination folder is no longer available - EX-2
    DocumentService-->DocumentController: MSG179 - The destination folder is not available.
    DocumentController-->LibraryScreen: MSG179 - The destination folder is not available.
    LibraryScreen-->Actor: MSG179 - The destination folder is not available.
else Any file is invalid or over the size limit - EX-1
    DocumentService-->DocumentController: MSG164 - Select valid PDF or DOCX files, each no larger than 10 MB.
    DocumentController-->LibraryScreen: MSG164 - Select valid PDF or DOCX files, each no larger than 10 MB.
    LibraryScreen-->Actor: MSG164 - Select valid PDF or DOCX files, each no larger than 10 MB.
else Every file is valid
    DocumentService->DocumentRepository: Get the total size of stored files
    activate DocumentRepository
    DocumentRepository->Database: Sum the sizes of stored files
    activate Database
    Database-->DocumentRepository: Total size
    deactivate Database
    DocumentRepository-->DocumentService: Total size
    deactivate DocumentRepository

    alt System storage is full - EX-3
        DocumentService-->DocumentController: MSG346 - The system's document storage is full.
        DocumentController-->LibraryScreen: MSG346 - The system's document storage is full.
        LibraryScreen-->Actor: MSG346 - The system's document storage is full.
    else Storage is available
        DocumentService->FileStorage: Store each file
        activate FileStorage
        FileStorage-->DocumentService: Files stored
        deactivate FileStorage

        DocumentService->DocumentRepository: Save the library documents
        activate DocumentRepository
        DocumentRepository->Database: Insert the library documents
        activate Database
        Database-->DocumentRepository: Saved
        deactivate Database
        DocumentRepository-->DocumentService: Saved
        deactivate DocumentRepository

        DocumentService-->DocumentController: Number of documents added
        DocumentController-->LibraryScreen: Number of documents added
        LibraryScreen-->Actor: MSG177 - Documents uploaded to your personal library.
    end
end

deactivate DocumentService
deactivate DocumentController
deactivate LibraryScreen
