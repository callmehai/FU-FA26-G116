title UC-86c - Play Collected Voice Sample

actor "Secretary" as Secretary
boundary "Meeting Detail Page" as MeetingPage
control ":Voice Collection Controller" as CollectionController
control ":Voice Collection Service" as CollectionService
participant ":Meeting Repository" as MeetingRepository
participant ":Voice ID Repository" as VoiceRepository
participant ":Voiceprint Protector" as Protector
participant ":Audit Service" as AuditService
database ":Database" as Database

Secretary->MeetingPage: Play a sample recorded at this Meeting
activate MeetingPage

MeetingPage->CollectionController: Get the collected sample recording
activate CollectionController

CollectionController->CollectionService: audio(meetingId, actorId, sampleId)
activate CollectionService

CollectionService->MeetingRepository: Get the Meeting with its current Secretary (BR-450, BR-451)
activate MeetingRepository
MeetingRepository->Database: Retrieve meeting with participants
activate Database
Database-->MeetingRepository: Meeting with participants / Not found
deactivate Database
MeetingRepository-->CollectionService: Meeting with participants / Not found
deactivate MeetingRepository

alt No access: not the current Secretary, or the Meeting fails its conditions - EX-1
    CollectionService-->CollectionController: Not found
    deactivate CollectionService

    CollectionController-->MeetingPage: Return failure
    deactivate CollectionController

    MeetingPage-->Secretary: Show the Meeting as not found
else Access granted
    CollectionService->VoiceRepository: Get the sample recorded at this Meeting by a Secretary (BR-452)
    activate VoiceRepository
    VoiceRepository->Database: Retrieve proxy sample of this Meeting
    activate Database
    Database-->VoiceRepository: Sample / None
    deactivate Database
    VoiceRepository-->CollectionService: Sample / None
    deactivate VoiceRepository

    alt Sample was recorded at another Meeting, or recorded by the person - EX-1
        CollectionService-->CollectionController: Not found
        deactivate CollectionService

        CollectionController-->MeetingPage: Return failure
        deactivate CollectionController

        MeetingPage-->Secretary: Show the sample as not found
    else Sample recorded at this Meeting
        CollectionService->Protector: Unprotect the sample recording (BR-364)
        activate Protector
        Protector-->CollectionService: Sample recording / Cannot be unprotected
        deactivate Protector

        alt Recording cannot be unprotected
            CollectionService-->CollectionController: No sample recording
            deactivate CollectionService

            CollectionController-->MeetingPage: No sample recording
            deactivate CollectionController

            MeetingPage-->Secretary: Show the sample as not playable
        else Recording available
            CollectionService->AuditService: Record the playback of someone else's sample (BR-453)
            activate AuditService
            AuditService-->CollectionService: Entry saved
            deactivate AuditService

            CollectionService-->CollectionController: Sample recording, not cached
            deactivate CollectionService

            CollectionController-->MeetingPage: Sample recording, not cached
            deactivate CollectionController

            MeetingPage-->Secretary: Play the sample
        end
    end
end

deactivate MeetingPage
