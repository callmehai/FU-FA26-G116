title UC-86c - Play Collected Voice Sample

actor "VoiceCollector" as Collector
boundary "Users Page" as UsersPage
control ":Voice Collection Controller" as CollectionController
control ":Voice Collection Service" as CollectionService
participant ":Voice ID Repository" as VoiceRepository
participant ":Voiceprint Protector" as Protector
participant ":Audit Service" as AuditService
database ":Database" as Database

Collector->UsersPage: Play a sample collected for the person
activate UsersPage

UsersPage->CollectionController: Get the collected sample recording
activate CollectionController

CollectionController->CollectionService: audio(userId, actorId, sampleId)
activate CollectionService

alt No access: no Voice sample collection permission - EX-1
    CollectionService-->CollectionController: Not found
    deactivate CollectionService

    CollectionController-->UsersPage: Return failure
    deactivate CollectionController

    UsersPage-->Collector: Show the sample as not found
else Access granted
    CollectionService->VoiceRepository: Get the sample collected for the person (BR-452)
    activate VoiceRepository
    VoiceRepository->Database: Retrieve collected sample of the person
    activate Database
    Database-->VoiceRepository: Sample / None
    deactivate Database
    VoiceRepository-->CollectionService: Sample / None
    deactivate VoiceRepository

    alt Sample does not exist, or was recorded by the person - EX-1
        CollectionService-->CollectionController: Not found
        deactivate CollectionService

        CollectionController-->UsersPage: Return failure
        deactivate CollectionController

        UsersPage-->Collector: Show the sample as not found
    else Collected sample exists
        CollectionService->Protector: Unprotect the sample recording (BR-364)
        activate Protector
        Protector-->CollectionService: Sample recording / Cannot be unprotected
        deactivate Protector

        alt Recording cannot be unprotected
            CollectionService-->CollectionController: No sample recording
            deactivate CollectionService

            CollectionController-->UsersPage: No sample recording
            deactivate CollectionController

            UsersPage-->Collector: Show the sample as not playable
        else Recording available
            CollectionService->AuditService: Record the playback of someone else's sample (BR-453)
            activate AuditService
            AuditService-->CollectionService: Entry saved
            deactivate AuditService

            CollectionService-->CollectionController: Sample recording, not cached
            deactivate CollectionService

            CollectionController-->UsersPage: Sample recording, not cached
            deactivate CollectionController

            UsersPage-->Collector: Play the sample
        end
    end
end

deactivate UsersPage
