title UC-56c - Record Voice Sample

actor "Internal User" as Actor
boundary "My Voice Page" as VoicePage
control ":Voice ID Controller" as VoiceController
control ":Voice ID Service" as VoiceService
participant ":Voice ID Repository" as VoiceRepository
participant ":Speech Recognition Service" as SpeechEngine
participant ":Voiceprint Protector" as Protector
participant ":Audit Service" as AuditService
database ":Database" as Database

Actor->VoicePage: Stop recording the sample and save it
activate VoicePage

VoicePage->VoiceController: Save the voice sample
activate VoiceController

VoiceController->VoiceService: addSample(actorId, audio, contentType)
activate VoiceService

VoiceService->VoiceRepository: Get the actor's consent and samples
activate VoiceRepository
VoiceRepository->Database: Retrieve consent and samples
activate Database
Database-->VoiceRepository: Consent and samples
deactivate Database
VoiceRepository-->VoiceService: Consent and samples
deactivate VoiceRepository

alt No standing consent - EX-1
    VoiceService-->VoiceController: MSG308 - Voice identification consent is required.
    VoiceController-->VoicePage: MSG308 - Voice identification consent is required.
    VoicePage-->Actor: MSG308 - Voice identification consent is required.
else Sample limit reached - EX-3
    VoiceService-->VoiceController: MSG310 - You have reached the sample limit.
    VoiceController-->VoicePage: MSG310 - You have reached the sample limit.
    VoicePage-->Actor: MSG310 - You have reached the sample limit.
else Sample may be added
    VoiceService->SpeechEngine: Check the sample and make its voiceprint
    activate SpeechEngine
    SpeechEngine-->VoiceService: Voiceprint / Sample rejected
    deactivate SpeechEngine

    alt Recording fails the quality rules - EX-2
        VoiceService-->VoiceController: MSG309 - Voice sample did not meet quality requirements.
        VoiceController-->VoicePage: MSG309 - Voice sample did not meet quality requirements.
        VoicePage-->Actor: MSG309 - Voice sample did not meet quality requirements.
    else Audio type is not supported - EX-2
        VoiceService-->VoiceController: MSG292 - Unsupported audio format.
        VoiceController-->VoicePage: MSG292 - Unsupported audio format.
        VoicePage-->Actor: MSG292 - Unsupported audio format.
    else Voiceprint made
        VoiceService->Protector: Protect the voiceprint
        activate Protector
        Protector-->VoiceService: Protected voiceprint
        deactivate Protector

        VoiceService->AuditService: Record the new sample
        activate AuditService
        AuditService-->VoiceService: Entry saved
        deactivate AuditService

        VoiceService->VoiceRepository: Save the sample under the consent
        activate VoiceRepository
        VoiceRepository->Database: Insert voice sample
        activate Database
        Database-->VoiceRepository: Saved
        deactivate Database
        VoiceRepository-->VoiceService: Saved
        deactivate VoiceRepository

        VoiceService-->VoiceController: MSG307 - Voice sample saved.
        VoiceController-->VoicePage: MSG307 - Voice sample saved.
        VoicePage-->Actor: MSG307 - Voice sample saved.
    end
end

deactivate VoiceService
deactivate VoiceController
deactivate VoicePage
