title UC-29 - View Meeting Participants

actor "User" as User
boundary "Participants Page" as ParticipantsPage
control ":Participant Controller" as ParticipantController
control ":Participant Service" as ParticipantService
participant ":Participant Repository" as ParticipantRepository
database ":Database" as Database

User->ParticipantsPage: Open participants
activate ParticipantsPage

ParticipantsPage->ParticipantController: Get participants (meetingId)
activate ParticipantController

ParticipantController->ParticipantService: getParticipants(meetingId, actor)
activate ParticipantService

ParticipantService->ParticipantRepository: Find meeting and check access
activate ParticipantRepository
ParticipantRepository->Database: Retrieve roster and access
activate Database
Database-->ParticipantRepository: Roster / not found
deactivate Database
ParticipantRepository-->ParticipantService: Roster / not found
deactivate ParticipantRepository

alt Roster is not accessible - EX-1
    ParticipantService-->ParticipantController: Not accessible
    ParticipantController-->ParticipantsPage: Return not found
    ParticipantsPage-->User: Roster not accessible

else Accessible
    ParticipantService-->ParticipantController: Roster
    ParticipantController-->ParticipantsPage: Roster
    ParticipantsPage-->User: Show roster (BR-016, BR-017)
end

deactivate ParticipantService
deactivate ParticipantController
deactivate ParticipantsPage
