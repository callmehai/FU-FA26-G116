title UC-08 - Assign System Role to User

actor "System Admin" as Admin
boundary "User Management Page" as UserPage
control ":User Controller" as UserController
control ":User Service" as UserService
participant ":User Repository" as UserRepository
participant ":Role Repository" as RoleRepository
participant ":Audit Log Repository" as AuditLogRepository
database ":Database" as Database

Admin->UserPage: Confirm system role assignment
activate UserPage
UserPage->UserController: Assign system role
activate UserController
UserController->UserService: assignSystemRole(userId, roleId, actorId)
activate UserService

UserService->UserRepository: Get target user and current role
activate UserRepository
UserRepository->Database: Retrieve target user and current role
activate Database
Database-->UserRepository: User and current role
deactivate Database
UserRepository-->UserService: User and current role
deactivate UserRepository

UserService->RoleRepository: Get selected system role
activate RoleRepository
RoleRepository->Database: Retrieve selected role
activate Database
Database-->RoleRepository: Role / Role unavailable
deactivate Database
RoleRepository-->UserService: Role / Role unavailable
deactivate RoleRepository
UserService->UserService: Validate target account and selected role

alt Selected role is not available - EX-1
    UserService-->UserController: MSG09 - Invalid account information.\nThe role does not exist
    deactivate UserService
    UserController-->UserPage: Return role validation error
    deactivate UserController
    UserPage-->Admin: MSG09 - Invalid account information.\nThe role does not exist

else Selected role is available
    activate UserController
    activate UserService
    UserService->UserRepository: Verify active account retains user administration permission
    activate UserRepository
    UserRepository->Database: Check active holders of user administration permission
    activate Database
    Database-->UserRepository: Remaining active holder count
    deactivate Database
    UserRepository-->UserService: Remaining active holder count
    deactivate UserRepository

    alt Assignment would remove the last active administrator - EX-2
        UserService-->UserController: MSG12 - Unable to change the status of this user account
        deactivate UserService
        UserController-->UserPage: Return role assignment conflict
        deactivate UserController
        UserPage-->Admin: MSG12 - Unable to change the status of this user account

    else Assignment is permitted
        activate UserController
        activate UserService
        UserService->UserRepository: Assign selected system role to user
        activate UserRepository
        UserRepository->Database: Save user role assignment
        activate Database
        Database-->UserRepository: Role assignment saved
        deactivate Database
        UserRepository-->UserService: Updated user
        deactivate UserRepository

        UserService->AuditLogRepository: Record system role assignment or change
        activate AuditLogRepository
        AuditLogRepository->Database: Insert audit record
        activate Database
        Database-->AuditLogRepository: Audit record saved
        deactivate Database
        AuditLogRepository-->UserService: Audit record saved
        deactivate AuditLogRepository

        UserService->UserService: Notify the user that their access changed so their sessions\napply the new role's permissions without signing in again (BR-314)
        UserService-->UserController: User role updated successfully
        deactivate UserService
        UserController-->UserPage: MSG10 - User account updated successfully
        deactivate UserController
        UserPage-->Admin: MSG10 - User account updated successfully
    end
end

deactivate UserPage
