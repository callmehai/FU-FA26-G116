title UC-08 - Reset User Password

actor "System Admin" as Admin
boundary "User Management Page" as UserPage
control ":User Controller" as UserController
control ":User Service" as UserService
participant ":User Repository" as UserRepository
participant ":Password Reset Token Store" as ResetTokenStore
control ":Email Service" as EmailService
control ":Audit Service" as AuditService
database ":Database" as Database

Admin->UserPage: Confirm password reset
activate UserPage

UserPage->UserController: Reset user password
activate UserController

UserController->UserService: requestPasswordReset(userId, adminId)
activate UserService

UserService->UserRepository: Get user account
activate UserRepository

UserRepository->Database: Retrieve user account
activate Database

Database-->UserRepository: User account
deactivate Database

UserRepository-->UserService: User account
deactivate UserRepository

alt User account no longer exists - EX-3
    UserService-->UserController: User account not found
    deactivate UserService

    UserController-->UserPage: Return not found
    deactivate UserController

    UserPage-->Admin: Inform that the account no longer exists\nand refresh the User Accounts list

else User account exists
    UserService->UserService: Check eligibility: account is active\nand signs in with a password (BR-22)

    alt Account is deactivated or has no password - EX-1
        UserService-->UserController: MSG14 - Password reset is not available\nfor this account
        deactivate UserService

        UserController-->UserPage: Return failure
        deactivate UserController

        UserPage-->Admin: MSG14 - Password reset is not available\nfor this account

    else Account is eligible
        UserService->ResetTokenStore: Create password reset request\n(60-minute, single-use link) (BR-20)
        activate ResetTokenStore

        ResetTokenStore-->UserService: Reset link, not usable yet
        deactivate ResetTokenStore

        UserService->EmailService: Send password reset email
        activate EmailService

        alt Email cannot be sent - EX-2
            EmailService-->UserService: Email delivery failed
            deactivate EmailService

            note over UserService, ResetTokenStore: The new link is never made usable;\na link sent earlier keeps working (BR-20)

            UserService-->UserController: MSG15 - Unable to send the password reset email.\nPlease try again.
            deactivate UserService

            UserController-->UserPage: Return failure
            deactivate UserController

            UserPage-->Admin: MSG15 - Unable to send the password reset email.\nPlease try again.

        else Email sent successfully
            EmailService-->UserService: Password reset email sent
            deactivate EmailService

            UserService->ResetTokenStore: Make the new link usable and\nretire the previous link (BR-20)
            activate ResetTokenStore

            ResetTokenStore-->UserService: Link activated
            deactivate ResetTokenStore

            UserService->AuditService: Record password reset request\n(actor: System Admin) (BR-16)
            activate AuditService

            AuditService->Database: Store audit record
            activate Database

            Database-->AuditService: Audit record stored
            deactivate Database

            AuditService-->UserService: Password reset request recorded
            deactivate AuditService

            UserService-->UserController: Password reset email sent
            deactivate UserService

            UserController-->UserPage: MSG13 - Password reset email sent successfully
            deactivate UserController

            UserPage-->Admin: MSG13 - Password reset email sent successfully
        end
    end
end

deactivate UserPage
