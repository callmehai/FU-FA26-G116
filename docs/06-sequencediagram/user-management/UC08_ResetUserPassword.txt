title UC-08 - Reset User Password

actor "System Admin" as Admin
boundary "User Management Page" as UserPage
control ":User Controller" as UserController
control ":User Service" as UserService
participant ":User Repository" as UserRepository
control ":Authentication Service" as AuthenticationService
control ":Email Service" as EmailService
control ":Audit Service" as AuditService
database ":Database" as Database

Admin->UserPage: Select Reset Password
UserPage-->Admin: Request confirmation
Admin->UserPage: Confirm password reset
UserPage->UserController: Reset user password
UserController->UserService: Initiate password reset
activate UserService
UserService->UserRepository: Get user account information
activate UserRepository
UserRepository->Database: Retrieve user account
activate Database
Database-->UserRepository: User account information
UserRepository-->UserService: User account information
alt User account is not eligible for password reset
    UserService-->UserController: Password reset unavailable
    UserController-->UserPage: MSG14 - Password reset is not available
    UserPage-->Admin: Display unavailable message
else User account is eligible
    UserService->AuthenticationService: Create password reset request
    activate AuthenticationService
    AuthenticationService-->UserService: Password reset request created
    UserService->EmailService: Send password reset email
    activate EmailService
    alt Email cannot be sent
        EmailService-->UserService: Email delivery failed
        UserService-->UserController: Password reset email failed
        UserController-->UserPage: MSG15 - Unable to send reset email
        UserPage-->Admin: Display error message
    else Email sent successfully
        EmailService-->UserService: Password reset email sent
        UserService->AuditService: Record password reset request
        activate AuditService
        AuditService->Database: Store audit record
        Database-->AuditService: Audit record created
        AuditService-->UserService: Password reset recorded
        UserService-->UserController: Password reset initiated successfully
        UserController-->UserPage: MSG13 - Password reset email sent successfully
        UserPage-->Admin: Display success message
        deactivate AuditService
    end
    deactivate EmailService
    deactivate AuthenticationService
end
deactivate Database
deactivate UserRepository
deactivate UserService
